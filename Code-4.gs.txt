/**
 * Optimized doPost - Uses action lookup table instead of switch statement
 */
function doPost(e) {
  try {
    var request = JSON.parse(e.postData.contents);
    var action = request.action;
    
    log_('📥 Action: ' + action);
    
    // Action handler lookup table (faster than switch for many cases)
    var handlers = {
      // MedWard Pro
      'analyzeLabsEnhanced': medward_analyzeLabsWithClaude,
      'analyzeLabs': medward_analyzeLabsWithClaude,
      'getDrugInfo': medward_getDrugInfo,
      'identifyMedication': medward_identifyMedication,
      'extractPatients': medward_extractPatients,
      'analyzeMeds': medward_analyzeMeds,
      'analyzeDocument': medward_analyzeDocument,
      'analyzeClinicalDocument': medward_analyzeClinicalDocument,
      'askClinical': medward_askClinical,
      'saveData': medward_saveData,
      'loadData': medward_loadData,
      
      // OnCall Assistant
      'oncallAskClinical': oncall_askClinical,
      'oncallDifferential': oncall_generateDifferential,
      'oncallTreatment': oncall_getTreatmentPlan,
      'oncallDrugInteraction': oncall_checkDrugInteractions,
      'oncallVerifyElectrolyte': oncall_verifyElectrolyteCorrection,
      'oncallVerifyVent': oncall_verifyVentilatorSettings,
      'oncallGetProtocol': oncall_getProtocol,
      'oncallGetReferenceRanges': oncall_getReferenceRanges
    };
    
    var response;
    
    if (action === 'ping') {
      response = {
        success: true,
        service: 'Unified Medical Backend',
        version: UNIFIED_CONFIG.VERSION,
        components: { medward: UNIFIED_CONFIG.MEDWARD_VERSION, oncall: UNIFIED_CONFIG.ONCALL_VERSION },
        timestamp: new Date().toISOString()
      };
    } else if (handlers[action]) {
      response = handlers[action](request);
    } else {
      response = { 
        success: false, 
        error: 'Unknown action: ' + action,
        availableActions: Object.keys(handlers).concat(['ping'])
      };
    }
    
    // Direct stringify - removed redundant parse verification
    return ContentService
      .createTextOutput(JSON.stringify(response))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    log_('❌ doPost Error: ' + error.toString());
    return ContentService
      .createTextOutput(JSON.stringify({ 
        success: false, 
        error: error.toString(),
        timestamp: new Date().toISOString()
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Unified media type extraction (eliminates code duplication)
 */
function extractMediaType_(base64Image) {
  var mediaType = 'image/jpeg';
  var rawBase64 = base64Image;
  
  if (base64Image.indexOf('data:') === 0) {
    var mediaMatch = base64Image.match(/^data:([^;]+);base64,/);
    if (mediaMatch) {
      mediaType = mediaMatch[1];
      rawBase64 = base64Image.split(',')[1];
    }
  } else {
    // Detect from base64 header signatures
    if (rawBase64.startsWith('/9j/')) mediaType = 'image/jpeg';
    else if (rawBase64.startsWith('iVBOR')) mediaType = 'image/png';
    else if (rawBase64.startsWith('R0lGOD')) mediaType = 'image/gif';
    else if (rawBase64.startsWith('UklGR')) mediaType = 'image/webp';
  }
  
  return { mediaType: mediaType, data: rawBase64 };
}

/**
 * Core Claude API caller - optimized with cached API key
 */
function callClaude_(systemPrompt, userMessage, model) {
  var apiKey = getApiKey_();
  
  if (!apiKey) {
    throw new Error('API key not configured. Add CLAUDE_API_KEY to Script Properties.');
  }
  
  if (!userMessage || typeof userMessage !== 'string' || userMessage.trim() === '') {
    throw new Error('User message is required and must be a non-empty string');
  }
  
  var payload = {
    model: model || UNIFIED_CONFIG.CLAUDE.MODEL,
    max_tokens: UNIFIED_CONFIG.CLAUDE.MAX_TOKENS,
    messages: [{ role: 'user', content: userMessage.trim() }]
  };
  
  if (systemPrompt && systemPrompt.trim()) {
    payload.system = systemPrompt.trim();
  }
  
  var response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    contentType: 'application/json',
    headers: {
      'x-api-key': apiKey.trim(),
      'anthropic-version': '2023-06-01'
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  
  var responseCode = response.getResponseCode();
  var responseText = response.getContentText();
  
  if (responseCode !== 200) {
    throw new Error('Claude API error (' + responseCode + '): ' + responseText);
  }
  
  var result = JSON.parse(responseText);
  
  if (!result.content || !result.content[0] || !result.content[0].text) {
    throw new Error('Unexpected response format from Claude');
  }
  
  return result.content[0].text;
}

/**
 * Claude with single image - uses unified media type extraction
 */
function callClaudeWithImage_(base64Image, prompt) {
  var apiKey = getApiKey_();
  if (!apiKey) throw new Error('API key not configured.');
  
  var imageData = extractMediaType_(base64Image);
  
  var response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    headers: {
      'x-api-key': apiKey.trim(),
      'anthropic-version': '2023-06-01',
      'content-type': 'application/json'
    },
    payload: JSON.stringify({
      model: UNIFIED_CONFIG.CLAUDE.MODEL,
      max_tokens: UNIFIED_CONFIG.CLAUDE.MAX_TOKENS,
      messages: [{
        role: 'user',
        content: [
          { type: 'image', source: { type: 'base64', media_type: imageData.mediaType, data: imageData.data } },
          { type: 'text', text: prompt }
        ]
      }]
    }),
    muteHttpExceptions: true
  });
  
  var result = JSON.parse(response.getContentText());
  if (result.error) throw new Error('Claude API Error: ' + result.error.message);
  return result.content[0].text;
}

/**
 * Claude with multiple images - uses unified media type extraction
 */
function callClaudeWithMultipleImages_(images, prompt) {
  var apiKey = getApiKey_();
  if (!apiKey) throw new Error('API key not configured.');
  
  var content = images.map(function(img) {
    var imageData = extractMediaType_(img);
    return { type: 'image', source: { type: 'base64', media_type: imageData.mediaType, data: imageData.data } };
  });
  
  content.push({ type: 'text', text: prompt });
  
  var response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    headers: {
      'x-api-key': apiKey.trim(),
      'anthropic-version': '2023-06-01',
      'content-type': 'application/json'
    },
    payload: JSON.stringify({
      model: UNIFIED_CONFIG.CLAUDE.MODEL,
      max_tokens: UNIFIED_CONFIG.CLAUDE.MAX_TOKENS,
      messages: [{ role: 'user', content: content }]
    }),
    muteHttpExceptions: true
  });
  
  var result = JSON.parse(response.getContentText());
  if (result.error) throw new Error('Claude API Error: ' + result.error.message);
  return result.content[0].text;
}

function medward_saveData(request) {
  try {
    var payload = request.payload || request.data || {};
    
    var dataToSave = {
      units: payload.units || [],
      patients: payload.patients || [],
      settings: payload.settings || {},
      unitRequests: payload.unitRequests || [],
      lastModified: new Date().toISOString(),
      modifiedBy: payload.modifiedBy || 'Unknown',
      version: Date.now()
    };
    
    var jsonStr = JSON.stringify(dataToSave);
    var sizeKB = Math.round(jsonStr.length / 1024);
    
    PropertiesService.getScriptProperties().setProperty('MEDWARD_DATA', jsonStr);
    
    return { 
      success: true,
      version: dataToSave.version,
      timestamp: dataToSave.lastModified,
      unitCount: dataToSave.units.length,
      patientCount: dataToSave.patients.length,
      sizeKB: sizeKB
    };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function medward_loadData() {
  try {
    var dataStr = PropertiesService.getScriptProperties().getProperty('MEDWARD_DATA');
    
    if (!dataStr) {
      return { 
        success: true, 
        data: { units: [], patients: [], settings: {}, unitRequests: [], lastModified: null, version: 0 }
      };
    }
    
    var data = JSON.parse(dataStr);
    
    return { 
      success: true, 
      data: data,
      version: data.version,
      lastModified: data.lastModified,
      unitCount: data.units ? data.units.length : 0,
      patientCount: data.patients ? data.patients.length : 0
    };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function medward_askClinical(request) {
  if (!request.question) return { success: false, error: 'No question provided' };
  
  try {
    var answer = callClaude_(SYSTEM_PROMPTS.MEDWARD_CLINICAL, request.question);
    return { success: true, answer: answer };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

/**
 * Drug info with CacheService (6-hour TTL)
 */
function medward_getDrugInfo(request) {
  if (!request.drugName) return { success: false, error: 'No drug name provided' };
  
  var cacheKey = 'DRUG_' + request.drugName.toLowerCase().replace(/\s+/g, '_');
  var cache = CacheService.getScriptCache();
  
  // Check cache first
  var cached = cache.get(cacheKey);
  if (cached) {
    try {
      return { success: true, drugInfo: JSON.parse(cached), cached: true };
    } catch (e) { /* Cache corrupted, fetch fresh */ }
  }
  
  try {
    var prompt = 'Provide clinical pharmacology information for: ' + request.drugName + '\n\n' +
      'Return ONLY a JSON object with this exact structure (no other text, no markdown):\n' +
      '{"drugClass":"","mechanism":"","indications":[],"dosing":"","sideEffects":[],"contraindications":[],"interactions":[],"monitoring":[],"warnings":[]}\n\n' +
      'Keep each array to 3-5 items maximum. Be concise. Start with { and end with }';
    
    var text = callClaude_('', prompt);
    text = text.trim().replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
    
    var jsonMatch = text.match(/\{[\s\S]*\}/);
    if (jsonMatch) text = jsonMatch[0];
    
    var drugInfo = JSON.parse(text);
    
    // Cache for 6 hours
    cache.put(cacheKey, JSON.stringify(drugInfo), UNIFIED_CONFIG.CACHE_TTL);
    
    return { success: true, drugInfo: drugInfo };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

/**
 * Optimized JSON cleaning (single pass regex)
 */
function cleanJsonResponse_(text) {
  return text
    .trim()
    .replace(/```json\n?|```\n?/g, '')
    .replace(/,(\s*[}\]])/g, '$1')
    .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F\uFEFF\u200B-\u200D\u2060]/g, '')
    .replace(/"([^"]*)\n([^"]*)"/g, '"$1 $2"');
}

/**
 * Recursive sanitizer (optimized)
 */
function sanitizeForJSON_(obj) {
  if (obj === null || obj === undefined) return obj;
  if (typeof obj === 'string') {
    return obj.replace(/[\x00-\x1F\x7F]/g, ' ').replace(/[\n\r\t]/g, ' ').trim();
  }
  if (Array.isArray(obj)) return obj.map(sanitizeForJSON_);
  if (typeof obj === 'object') {
    var result = {};
    for (var key in obj) {
      if (obj.hasOwnProperty(key)) result[key] = sanitizeForJSON_(obj[key]);
    }
    return result;
  }
  return obj;
}

function medward_analyzeLabsWithClaude(request) {
  try {
    // Extract images - early validation
    var images = [];
    if (request.images && Array.isArray(request.images)) {
      images = request.images.filter(function(img) { return img && typeof img === 'string' && img.length > 0; });
    } else if (request.image && typeof request.image === 'string' && request.image.length > 0) {
      images = [request.image];
    }
    
    if (images.length === 0) return { success: false, error: 'No valid image data provided' };
    
    var aiText = images.length === 1 
      ? callClaudeWithImage_(images[0], SYSTEM_PROMPTS.LAB_ANALYSIS)
      : callClaudeWithMultipleImages_(images, SYSTEM_PROMPTS.LAB_ANALYSIS);
    
    var cleanedText = cleanJsonResponse_(aiText);
    var jsonMatch = cleanedText.match(/\{[\s\S]*\}/);
    if (jsonMatch) cleanedText = jsonMatch[0];
    
    var parsed;
    try {
      parsed = JSON.parse(cleanedText);
    } catch (parseError) {
      // Attempt Claude-based repair
      log_('🔄 Attempting JSON repair...');
      try {
        var fixPrompt = 'Fix this broken JSON and return ONLY the corrected JSON object:\n\n' + cleanedText.substring(0, 15000);
        var fixedText = callClaude_(SYSTEM_PROMPTS.JSON_REPAIR, fixPrompt);
        fixedText = cleanJsonResponse_(fixedText);
        var fixedMatch = fixedText.match(/\{[\s\S]*\}/);
        if (fixedMatch) fixedText = fixedMatch[0];
        parsed = JSON.parse(fixedText);
      } catch (fixError) {
        return { 
          success: false, 
          error: 'JSON parsing failed: ' + parseError.message,
          debug: { rawResponseLength: aiText.length, rawResponsePreview: aiText.substring(0, 300) }
        };
      }
    }
    
    return {
      success: true,
      labData: sanitizeForJSON_(parsed.labData),
      confidence: parsed.confidence || 0.8,
      patientInfo: sanitizeForJSON_(parsed.patientInfo)
    };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

/**
 * Analyzes any clinical document - EMR screenshots, notes, labs, reports
 * Intelligently extracts patient info, diagnoses, medications, labs, findings
 */
function medward_analyzeDocument(request) {
  if (!request.image) return { success: false, error: 'No image/document provided' };
  
  try {
    log_('📄 Analyzing clinical document...');
    
    // Handle PDF vs image
    var mediaType = request.mediaType || null;
    var result;
    
    if (mediaType === 'application/pdf') {
      // For PDFs, use document type in Claude API
      result = callClaudeWithDocument_(request.image, 'application/pdf', SYSTEM_PROMPTS.DOCUMENT_ANALYZE);
    } else {
      result = callClaudeWithImage_(request.image, SYSTEM_PROMPTS.DOCUMENT_ANALYZE);
    }
    
    var cleanedText = cleanJsonResponse_(result);
    var jsonMatch = cleanedText.match(/\{[\s\S]*\}/);
    if (jsonMatch) cleanedText = jsonMatch[0];
    
    var parsed;
    try {
      parsed = JSON.parse(cleanedText);
    } catch (parseError) {
      var fixPrompt = 'Fix this broken JSON and return ONLY valid JSON:\n\n' + cleanedText.substring(0, 10000);
      var fixedText = callClaude_(SYSTEM_PROMPTS.JSON_REPAIR, fixPrompt);
      fixedText = cleanJsonResponse_(fixedText);
      var fixedMatch = fixedText.match(/\{[\s\S]*\}/);
      if (fixedMatch) fixedText = fixedMatch[0];
      parsed = JSON.parse(fixedText);
    }
    
    log_('✅ Document type: ' + (parsed.documentType || 'unknown') + ', Meds: ' + (parsed.medications ? parsed.medications.length : 0));
    
    return {
      success: true,
      documentType: parsed.documentType || 'other',
      patientInfo: parsed.patientInfo || null,
      date: parsed.date || null,
      diagnoses: parsed.diagnoses || [],
      medications: parsed.medications || [],
      vitals: parsed.vitals || null,
      labValues: parsed.labValues || [],
      planItems: parsed.planItems || [],
      findings: parsed.findings || {},
      allergies: parsed.allergies || [],
      rawNotes: parsed.rawNotes || null,
      confidence: parsed.confidence || 0.8,
      timestamp: new Date().toISOString()
    };
    
  } catch (error) {
    log_('❌ analyzeDocument error: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * Claude with PDF document
 */
function callClaudeWithDocument_(base64Data, mediaType, prompt) {
  var apiKey = getApiKey_();
  if (!apiKey) throw new Error('API key not configured.');
  
  var response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    headers: {
      'x-api-key': apiKey.trim(),
      'anthropic-version': '2023-06-01',
      'content-type': 'application/json'
    },
    payload: JSON.stringify({
      model: UNIFIED_CONFIG.CLAUDE.MODEL,
      max_tokens: UNIFIED_CONFIG.CLAUDE.MAX_TOKENS,
      messages: [{
        role: 'user',
        content: [
          { type: 'document', source: { type: 'base64', media_type: mediaType, data: base64Data } },
          { type: 'text', text: prompt }
        ]
      }]
    }),
    muteHttpExceptions: true
  });
  
  var result = JSON.parse(response.getContentText());
  if (result.error) throw new Error('Claude API Error: ' + result.error.message);
  return result.content[0].text;
}

/**
 * Identifies medication(s) from image - handles single drugs, med lists, prescriptions, lab reports
 * NEW in v5.0 - Camera-based medication lookup (supports multiple input types)
 */
function medward_identifyMedication(request) {
  if (!request.image) return { success: false, error: 'No image provided' };
  
  try {
    log_('📷 Analyzing image for medications...');
    
    // Step 1: Identify medications from the image
    var identifyResult = callClaudeWithImage_(request.image, SYSTEM_PROMPTS.MEDICATION_IDENTIFY);
    
    // Clean and parse the response
    var cleanedText = cleanJsonResponse_(identifyResult);
    var jsonMatch = cleanedText.match(/\{[\s\S]*\}/);
    if (jsonMatch) cleanedText = jsonMatch[0];
    
    var identified;
    try {
      identified = JSON.parse(cleanedText);
    } catch (parseError) {
      // Try to repair with Claude
      var fixPrompt = 'Fix this broken JSON and return ONLY valid JSON:\n\n' + cleanedText;
      var fixedText = callClaude_(SYSTEM_PROMPTS.JSON_REPAIR, fixPrompt);
      fixedText = cleanJsonResponse_(fixedText);
      var fixedMatch = fixedText.match(/\{[\s\S]*\}/);
      if (fixedMatch) fixedText = fixedMatch[0];
      identified = JSON.parse(fixedText);
    }
    
    // Normalize response - ensure medications is always an array
    var medications = identified.medications || [];
    
    // Handle legacy single-drug format (backward compatibility)
    if (medications.length === 0 && identified.drugName) {
      medications = [{
        drugName: identified.drugName,
        brandName: identified.brandName,
        strength: identified.strength,
        form: identified.form,
        frequency: null,
        route: null,
        indication: null
      }];
    }
    
    log_('✅ Found ' + medications.length + ' medication(s), type: ' + (identified.imageType || 'unknown'));
    
    // Step 2: If single drug with high confidence, auto-fetch drug info
    var drugInfo = null;
    if (medications.length === 1 && identified.confidence >= 0.7 && medications[0].drugName) {
      log_('📚 Fetching drug info for: ' + medications[0].drugName);
      var infoResult = medward_getDrugInfo({ drugName: medications[0].drugName });
      if (infoResult.success) {
        drugInfo = infoResult.drugInfo;
      }
    }
    
    return {
      success: true,
      imageType: identified.imageType || (medications.length > 1 ? 'medication_list' : 'single_drug'),
      medications: medications,
      confidence: identified.confidence || 0.8,
      notes: identified.notes || null,
      drugInfo: drugInfo, // Only populated for single drug lookups
      count: medications.length,
      message: identified.confidence < 0.7 ? 'Low confidence - please verify' : null,
      timestamp: new Date().toISOString()
    };
    
  } catch (error) {
    log_('❌ identifyMedication error: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

function oncall_verifyVentilatorSettings(request) {
  if (!request.height || !request.gender) {
    return { success: false, error: 'Height and gender required' };
  }
  
  try {
    var heightCm = parseFloat(request.height);
    var pbw = (request.gender.toLowerCase() === 'male')
      ? 50 + 0.91 * (heightCm - 152.4)
      : 45.5 + 0.91 * (heightCm - 152.4);
    pbw = Math.round(pbw * 10) / 10;
    
    var tvLow = Math.round(pbw * 6);
    var tvHigh = Math.round(pbw * 8);
    
    var prompt = 'Patient: ' + request.gender + ', ' + heightCm + ' cm\n' +
      'PBW: ' + pbw + ' kg\n' +
      'Recommended TV: ' + tvLow + '-' + tvHigh + ' mL\n' +
      'Proposed TV: ' + (request.tidalVolume || 'Not specified');
    
    return {
      success: true,
      verification: callClaude_(SYSTEM_PROMPTS.VENTILATOR, prompt),
      calculations: { pbw: pbw, tvRange: tvLow + '-' + tvHigh + ' mL' },
      timestamp: new Date().toISOString()
    };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function oncall_getProtocol(request) {
  if (!request.protocol) {
    return { success: true, availableProtocols: Object.keys(CLINICAL_PROTOCOLS) };
  }
  
  var key = request.protocol.toLowerCase();
  if (CLINICAL_PROTOCOLS[key]) {
    return { success: true, protocol: CLINICAL_PROTOCOLS[key], name: request.protocol };
  }
  
  return { success: false, error: 'Protocol not found', available: Object.keys(CLINICAL_PROTOCOLS) };
}

function oncall_getReferenceRanges(request) {
  if (request && request.parameter) {
    var key = request.parameter.toLowerCase().replace('+', '').replace(' ', '_');
    if (REFERENCE_RANGES[key]) {
      return { success: true, parameter: request.parameter, range: REFERENCE_RANGES[key] };
    }
    return { success: false, error: 'Parameter not found' };
  }
  
  return { success: true, ranges: REFERENCE_RANGES };
}
